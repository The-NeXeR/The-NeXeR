```c
#include <stdio.h>

typedef struct {
    const char *name, *alias, *role;
    const char *stack[4];
    const char *learning[3];
} Dev;

int main(void) {
    Dev me = {
        .name     = "Pouya Nasiri",
        .alias    = "The NeXeR",
        .role     = "Computer Engineering student",
        .stack    = { "C", "Flutter", "Dart", "SQLite" },
        .learning = { "systems programming", "network programming", "socket programming" },
    };

    printf("%s — %s\n%s\n\n", me.name, me.alias, me.role);

    printf("stack:\n");
    for (int i = 0; i < 4; i++) printf("  - %s\n", me.stack[i]);

    printf("currently learning:\n");
    for (int i = 0; i < 3; i++) printf("  - %s\n", me.learning[i]);

    puts("\nmaybe building something useful");
    return 0;
}
```
